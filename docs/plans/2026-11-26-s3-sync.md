# S3 Sync Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Sync claude-mem's memory DB, chroma index and configuration to an own S3-compatible bucket (Yandex Object Storage) on a schedule and on demand, and restore from it — the "Cloud Sync" capability without the hosted hub.

**Architecture:** A worker-side service (`src/services/s3sync/`) uploads gzip'd, manifest-described snapshot sets to S3 under `claude-mem-sync/<device>/…`, reusing the backup stack's verified `VACUUM INTO` snapshot path and the chroma writer lock. The CLI is a thin HTTP client to new worker routes, exactly like `backup`/`restore`. Restores stage through the existing boot-time `restore-pending` mechanism, plus a new boot-time chroma dir swap.

**Tech Stack:** TypeScript, Bun (`Bun.S3Client`, `bun:sqlite` in child jobs), express routes, `node:zlib` streaming gzip, bun test. No new npm dependencies.

**Design doc:** `docs/plans/2026-11-26-s3-sync-design.md` (approved). Read it first.

**Working agreements:**
- Branch: `feat/s3-sync`. Push to remote **`fork`** (`diarworld/claude-mem`), never to `origin` (upstream). No PRs upstream.
- The worker runs under Bun; the npx CLI runs under Node. Only the worker touches SQLite and S3.
- Never copy the live `claude-mem.db` raw. Never copy chroma outside the writer lock. Never upload logs or caches.
- No S3 calls in tests — the transport is an injected interface.

```ts
 * Push-pipeline phase of one S3 Sync set (docs/architecture/state-machines.md):

export const S3_SYNC_PUSH_PHASES = ['idle', 'snapshotting', 'uploading', 'ok', 'error'] as const;

  idle: 'Not started yet',
  snapshotting: 'Creating local snapshot',
  uploading: 'Uploading to S3',
  ok: 'Up to date',
  error: 'Last push failed',
};

export function isS3SyncPushPhase(value: unknown): value is S3SyncPushPhase {
  return typeof value === 'string' && (S3_SYNC_PUSH_PHASES as readonly string[]).includes(value);
}
```

```ts
/**
 * Object-key layout, generation manifests and pure helpers for S3 Sync.
 * Node-safe (no Bun imports): the CLI bundle imports this module too.
 *
 * Layout in the bucket:
 *   claude-mem-sync/<device>/<set>/LATEST            {"generation": "...", "createdAt": ...}
 *   claude-mem-sync/<device>/<set>/gen/<stamp>/manifest.json
 *   claude-mem-sync/<device>/<set>/gen/<stamp>/<relative keys…>
 */
import { createHash } from 'node:crypto';
import { createReadStream } from 'node:fs';

export const S3_SYNC_ROOT = 'claude-mem-sync';
export const S3_SYNC_SETS = ['db', 'chroma'] as const;
export type S3SyncSet = (typeof S3_SYNC_SETS)[number];

export interface S3SyncManifestFile {
  /** Key relative to the generation prefix. */
  key: string;
  bytes: number;
  sha256: string;
}

export interface S3SyncManifest {
  version: 1;
  set: S3SyncSet;
  generation: string;
  createdAt: number;
  device: string;
  /** chroma set only: the db generation this index is consistent with. */
  pairedDbGeneration?: string;
  files: S3SyncManifestFile[];
}

export function sanitizeDeviceName(name: string): string {
  const cleaned = name.toLowerCase().replace(/[^a-z0-9._-]+/g, '-').replace(/^-+|-+$/g, '');
  return cleaned || 'device';
}

/** Sanitized ISO stamps contain no ':' or '.', so key order is time order. */
export function generationStamp(ms: number): string {
  return new Date(ms).toISOString().replace(/[:.]/g, '-');
}

export function deviceRoot(device: string): string {
  return `${S3_SYNC_ROOT}/${sanitizeDeviceName(device)}`;
}

export function setPrefix(device: string, set: S3SyncSet): string {
  return `${deviceRoot(device)}/${set}`;
}

export function generationPrefix(device: string, set: S3SyncSet, generation: string): string {
  return `${setPrefix(device, set)}/gen/${generation}`;
}

export function manifestObjectKey(device: string, set: S3SyncSet, generation: string): string {
  return `${generationPrefix(device, set, generation)}/manifest.json`;
}

export function latestObjectKey(device: string, set: S3SyncSet): string {
  return `${setPrefix(device, set)}/LATEST`;
}

export function latestMarkerPayload(generation: string, createdAt: number): string {
  return JSON.stringify({ generation, createdAt });
}

export function parseLatestMarker(text: string): { generation: string; createdAt: number } | null {
  try {
    const parsed = JSON.parse(text) as { generation?: unknown; createdAt?: unknown };
    if (typeof parsed?.generation !== 'string' || parsed.generation === '') return null;
    return { generation: parsed.generation, createdAt: Number(parsed.createdAt) || 0 };
  } catch {
    return null;
  }
}

export async function sha256File(filePath: string): Promise<string> {
  return new Promise((resolve, reject) => {
    const hash = createHash('sha256');
    const stream = createReadStream(filePath);
    stream.on('data', chunk => hash.update(chunk));
    stream.on('error', reject);
    stream.on('end', () => resolve(hash.digest('hex')));
  });
}

/** Objects that changed since the previous manifest, by relative key + hash. */
export function changedFiles(previous: S3SyncManifest | null, next: S3SyncManifest): S3SyncManifestFile[] {
  if (!previous || previous.set !== next.set) return next.files;
  const before = new Map(previous.files.map(file => [file.key, file.sha256]));
  return next.files.filter(file => before.get(file.key) !== file.sha256);
}

/**
 * Generations to delete so `keep` newest survive. `freshGeneration` (the one
 * just uploaded) always survives even if the clock moved backward.
 */
export function pruneGenerations(keys: string[], set: S3SyncSet, keep: number, freshGeneration?: string): string[] {
  const gens = new Set<string>();
  for (const key of keys) {
    const parts = key.split('/');
    // [claude-mem-sync, <device>, <set>, gen, <stamp>, …] — device names are sanitized, no '/'
    if (parts.length >= 5 && parts[0] === S3_SYNC_ROOT && parts[2] === set && parts[3] === 'gen') {
      gens.add(parts[4]);
    }
  }
  const sorted = [...gens].sort().reverse();
  const survivors = new Set(sorted.slice(0, Math.max(1, keep)));
  if (freshGeneration) survivors.add(freshGeneration);
  return sorted.filter(gen => !survivors.has(gen));
}
```

```ts
/**
 * S3-compatible storage access for the worker (Bun runtime only). The
 * interface exists so tests inject an in-memory fake — no test ever touches
 * the network. Built and validated against Yandex Object Storage
 * (https://storage.yandexcloud.net, region ru-central1, path-style).
 */
import { createReadStream, createWriteStream, statSync } from 'node:fs';
import { pipeline } from 'node:stream/promises';

export interface S3Transport {
  upload(localPath: string, key: string): Promise<void>;
  download(key: string, localPath: string): Promise<void>;
  deleteObject(key: string): Promise<void>;
  listKeys(prefix: string): Promise<string[]>;
}

export interface BunS3TransportConfig {
  endpoint: string;
  bucket: string;
  region: string;
  accessKeyId: string;
  secretAccessKey: string;
}

/** Single-shot PUT up to this size; multipart above it. */
const MULTIPART_THRESHOLD_BYTES = 64 * 1024 * 1024;
const MULTIPART_PART_SIZE = 16 * 1024 * 1024;

export class BunS3Transport implements S3Transport {
  private readonly client: InstanceType<(typeof import('bun'))['S3Client']>;

  constructor(config: BunS3TransportConfig) {
    if (typeof Bun === 'undefined') {
      throw new Error('S3 transport needs the Bun runtime (the worker), not Node');
    }
    this.client = new Bun.S3Client({
      accessKeyId: config.accessKeyId,
      secretAccessKey: config.secretAccessKey,
      bucket: config.bucket,
      endpoint: config.endpoint,
      region: config.region,
    });
  }

  async upload(localPath: string, key: string): Promise<void> {
    const bytes = statSync(localPath).size;
    if (bytes <= MULTIPART_THRESHOLD_BYTES) {
      // Bun.file() is a Blob; s3.write accepts it without buffering the file
      // into memory (validated against Yandex: 200 OK).
      await this.client.write(key, Bun.file(localPath));
      return;
    }
    const writer = this.client.file(key).writer({ partSize: MULTIPART_PART_SIZE });
    const stream = Bun.file(localPath).stream();
    for await (const chunk of stream) {
      await writer.write(chunk as unknown as Buffer);
    }
    await writer.end();
  }

  async download(key: string, localPath: string): Promise<void> {
    await Bun.write(localPath, this.client.file(key));
  }

  async deleteObject(key: string): Promise<void> {
    await this.client.delete(key);
  }

  async listKeys(prefix: string): Promise<string[]> {
    const keys: string[] = [];
    let cursor: string | undefined;
    do {
      const page = await this.client.list({ prefix, cursor, maxKeys: 1000 });
      for (const entry of page.contents ?? []) keys.push(entry.key);
      cursor = page.nextCursor ?? undefined;
    } while (cursor);
    return keys;
  }
}
```

```ts
 * BackupManager — a live db is never copied raw. chroma.sqlite3 gets the same
 * treatment; the HNSW binary files are copied directly under the worker's
 * chroma writer lock with a stability check (size+mtime read before and after
 * the copy; a change means the index was written mid-copy → retry once, then
 * fail the chroma set; the db set is unaffected and chroma is rebuildable).

import { copyFileSync, existsSync, mkdirSync, readdirSync, statSync, unlinkSync } from 'node:fs';
import { createReadStream, createWriteStream } from 'node:fs';
import path from 'node:path';
import { pipeline } from 'node:stream/promises';
import { createGzip, createGunzip } from 'node:zlib';
import { runSqliteFileJob } from '../backup/sqlite-file-job.js';

export interface SnapshotInput {
  dbPath: string;
  reviewsDbPath: string;
  chromaDir: string;
  dataDir: string;
  settingsPath: string;
  envPath: string;
  stateDir: string;
}

export interface SnapshotEntry {
  /** File on disk (inside the work dir), ready to upload. */
  absolutePath: string;
  /** Key relative to the generation prefix. */
  key: string;
  bytes: number;
}

export async function gzipFile(src: string, dst: string): Promise<void> {
  await pipeline(createReadStream(src), createGzip({ level: 6 }), createWriteStream(dst));
}

export async function gunzipFile(src: string, dst: string): Promise<void> {
  await pipeline(createReadStream(src), createGunzip(), createWriteStream(dst));
}

async function snapshotAndGzip(source: string, workDir: string, key: string, signal?: AbortSignal): Promise<SnapshotEntry> {
  const raw = path.join(workDir, path.basename(key).replace(/\.gz$/, ''));
  const gz = path.join(workDir, key.split('/').pop()!);
  await runSqliteFileJob({ kind: 'snapshot', source, target: raw }, signal);
  await gzipFile(raw, gz);
  unlinkSync(raw);
  return { absolutePath: gz, key, bytes: statSync(gz).size };
}

function copyIfExists(src: string, key: string, workDir: string): SnapshotEntry | null {
  if (!existsSync(src)) return null;
  const dst = path.join(workDir, key);
  copyFileSync(src, dst);
  return { absolutePath: dst, key, bytes: statSync(dst).size };
}

export async function buildDbSnapshotSet(
  workDir: string,
  input: SnapshotInput,
  options: { includeEnv: boolean; signal?: AbortSignal },
): Promise<SnapshotEntry[]> {
  mkdirSync(workDir, { recursive: true });
  const entries: SnapshotEntry[] = [];
  entries.push(await snapshotAndGzip(input.dbPath, workDir, 'claude-mem.db.gz', options.signal));
  if (existsSync(input.reviewsDbPath)) {
    entries.push(await snapshotAndGzip(input.reviewsDbPath, workDir, 'work-view-reviews.db.gz', options.signal));
  }
  for (const [src, key] of [
    [input.settingsPath, 'settings.json'],
    [path.join(path.dirname(input.settingsPath), 'MY-RULES.md'), 'MY-RULES.md'],
    [path.join(path.dirname(input.settingsPath), 'chroma-sync-state.json'), 'chroma-sync-state.json'],
  ] as const) {
    const copied = copyIfExists(src, key, workDir);
    if (copied) entries.push(copied);
  }
  if (options.includeEnv) {
    const env = copyIfExists(input.envPath, 'env', workDir);
    if (env) entries.push(env);
  }
  if (existsSync(input.stateDir)) {
    for (const name of readdirSync(input.stateDir, { withFileTypes: true })) {
      if (!name.isFile() || !name.name.endsWith('.json')) continue;
      const copied = copyIfExists(path.join(input.stateDir, name.name), `state/${name.name}`, workDir);
      if (copied) entries.push(copied);
    }
  }
  return entries;
}

/** Copy an HNSW file, refusing a torn copy: size+mtime must be identical before and after. */
function copyStable(src: string, dst: string): void {
  for (let attempt = 0; attempt < 2; attempt++) {
    const before = statSync(src);
    copyFileSync(src, dst);
    const after = statSync(src);
    if (before.size === after.size && before.mtimeMs === after.mtimeMs) return;
    require('node:timers/promises').setTimeout(250);
  }
  throw new Error(`chroma index file changed during copy: ${src} (another writer? is the chroma writer lock held?)`);
}

export async function buildChromaSnapshotSet(
  workDir: string,
  input: SnapshotInput,
  options: { signal?: AbortSignal } = {},
): Promise<SnapshotEntry[]> {
  mkdirSync(workDir, { recursive: true });
  const chromaDbPath = path.join(input.chromaDir, 'chroma.sqlite3');
  if (!existsSync(chromaDbPath)) throw new Error(`chroma.sqlite3 not found at ${chromaDbPath}`);
  const entries: SnapshotEntry[] = [];
  entries.push(await snapshotAndGzip(chromaDbPath, workDir, 'chroma/chroma.sqlite3.gz', options.signal));
  for (const entry of readdirSync(input.chromaDir, { withFileTypes: true })) {
    if (!entry.isDirectory()) continue;
    for (const file of readdirSync(path.join(input.chromaDir, entry.name), { withFileTypes: true })) {
      if (!file.isFile()) continue;
      const relKey = `chroma/${entry.name}/${file.name}`;
      const dst = path.join(workDir, relKey);
      mkdirSync(path.dirname(dst), { recursive: true });
      copyStable(path.join(input.chromaDir, entry.name, file.name), dst);
      entries.push({ absolutePath: dst, key: relKey, bytes: statSync(dst).size });
    }
  }
  return entries;
}
```

```ts
/**
 * Boot-time chroma directory restore, staged by an S3 Sync pull. Runs in
 * worker-service.ts right after applyPendingRestore, before ChromaMcpManager
 * is constructed and before anything opens the vector store. Apply-once like
 * the database restore: the marker is cleared as soon as it is read. The
 * previous chroma dir is renamed aside as chroma-pre-restore-<ts> and kept.
 */
import { existsSync, renameSync } from 'node:fs';
import path from 'node:path';
import { logger } from '../../utils/logger.js';
import { renameWithWindowsRetry } from '../../shared/atomic-json.js';
import { clearPendingChromaRestore, readPendingChromaRestore } from '../backup/snapshot-files.js';

export interface ChromaRestoreResult {
  ok: boolean;
  error: string | null;
  preRestoreDir: string | null;
}

export function applyPendingChromaRestore(chromaDir: string, markerPath = paths.pendingChromaRestore()): ChromaRestoreResult | null {
  let pending;
  try {
    pending = readPendingChromaRestore(markerPath);
  } catch (error) {
    logger.error('S3SYNC', 'Chroma restore marker is unreadable; leaving chroma untouched', { markerPath }, error instanceof Error ? error : new Error(String(error)));
    return { ok: false, error: (error as Error).message, preRestoreDir: null };
  }
  if (pending === null) return null;
  clearPendingChromaRestore(markerPath);
  const preRestoreDir = path.join(path.dirname(chromaDir), `chroma-pre-restore-${new Date().toISOString().replace(/[:.]/g, '-')}`);
  try {
    if (!existsSync(pending.stagingDir)) throw new Error(`chroma staging dir is missing: ${pending.stagingDir}`);
    if (existsSync(chromaDir)) renameSync(chromaDir, preRestoreDir);
    try {
      renameWithWindowsRetry(pending.stagingDir, chromaDir);
    } catch (error) {
      // put the old dir back; nothing was overwritten
      if (existsSync(preRestoreDir)) renameSync(preRestoreDir, chromaDir);
      throw error;
    }
    logger.info('S3SYNC', 'Chroma directory restored from S3 staging', { stagingDir: pending.stagingDir, preRestoreDir, pairedDbGeneration: pending.pairedDbGeneration });
    return { ok: true, error: null, preRestoreDir };
  } catch (error) {
    const normalized = error instanceof Error ? error : new Error(String(error));
    logger.error('S3SYNC', 'Chroma restore FAILED; the worker starts with the unchanged chroma directory', { stagingDir: pending.stagingDir }, normalized);
    return { ok: false, error: normalized.message, preRestoreDir: null };
  }
}
```

```ts
import { logger } from '../../../../utils/logger.js';
import { BaseRouteHandler } from '../BaseRouteHandler.js';
import type { DatabaseManager } from '../../DatabaseManager.js';
import { isForeignLoopbackBrowserWrite } from './SettingsRoutes.js';
import { S3_SYNC_SETS } from '../../../../services/s3sync/manifest.js';

/**
 * S3 Sync endpoints. Registered unconditionally (BackupRoutes pattern): a
 * disabled install answers status with {configured: false}.
 *
 * GET  /api/s3sync/status - manager status (or {configured: false})
 * POST /api/s3sync/push   - {set?: 'db'|'chroma'|'all'} push now (409 if busy)
 * POST /api/s3sync/pull   - {set?, generation?, force?} download + verify + stage
 *
 * There is no restore endpoint: pull stages, and the next worker boot applies
 * the database (restore-swap) and the chroma dir (chroma-restore).
 */
export class S3SyncRoutes extends BaseRouteHandler {
  constructor(
    private dbManager: DatabaseManager,
  ) {
    super();
  }

  setupRoutes(app: express.Application): void {
    app.get('/api/s3sync/status', this.handleGetStatus.bind(this));
    app.post('/api/s3sync/push', this.handlePush.bind(this));
    app.post('/api/s3sync/pull', this.handlePull.bind(this));
  }

  private parseSet(body: unknown): 'db' | 'chroma' | 'all' {
    const set = (body as { set?: string } | undefined)?.set ?? 'all';
    if (set === 'all' || (S3_SYNC_SETS as readonly string[]).includes(set)) return set as 'db' | 'chroma' | 'all';
    throw new Error(`invalid set: ${set}`);
  }

  private handleGetStatus = this.wrapHandler(async (_req: Request, res: Response): Promise<void> => {
    const manager = this.dbManager.getS3SyncManager();
    if (!manager) {
      res.json({ configured: false });
      return;
    }
    res.json(manager.status());
  });

  private handlePush = this.wrapHandler(async (req: Request, res: Response): Promise<void> => {
    if (isForeignLoopbackBrowserWrite(req)) {
      res.status(403).json({ error: 'S3 Sync pushes from a different localhost origin are not allowed' });
      return;
    }
    const manager = this.dbManager.getS3SyncManager();
    if (!manager) {
      res.status(400).json({ error: 'S3 Sync is not enabled (set CLAUDE_MEM_S3_SYNC_ENABLED to "true", add the bucket keys to .env, and restart the worker)' });
      return;
    }
    let set: 'db' | 'chroma' | 'all';
    try { set = this.parseSet(req.body); } catch (error) {
      res.status(400).json({ error: (error as Error).message });
      return;
    }
    const outcome = await manager.runNow(set);
    if (outcome === null) {
      res.status(409).json({ error: 'An S3 Sync push is already running' });
      return;
    }
    res.json({ success: true, outcome });
  });

  private handlePull = this.wrapHandler(async (req: Request, res: Response): Promise<void> => {
    if (isForeignLoopbackBrowserWrite(req)) {
      res.status(403).json({ error: 'S3 Sync pulls from a different localhost origin are not allowed' });
      return;
    }
    const manager = this.dbManager.getS3SyncManager();
    if (!manager) {
      res.status(400).json({ error: 'S3 Sync is not enabled' });
      return;
    }
    const body = (req.body ?? {}) as { set?: string; generation?: string; force?: boolean };
    try {
      const outcome = await manager.pull({
        set: this.parseSet(body),
        generation: typeof body.generation === 'string' && body.generation ? body.generation : undefined,
        force: body.force === true,
      });
      res.json({ success: true, outcome, note: 'Restart the worker to apply the staged restore (claude-mem s3sync pull restarts it for you).' });
    } catch (error) {
      logger.warn('S3SYNC', 'Pull failed', { set: body.set }, error instanceof Error ? error : new Error(String(error)));
      res.status(400).json({ error: (error as Error).message });
    }
  });
}
```
